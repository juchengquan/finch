```swift
// Selectors/SelectorFixture.swift — the JSON shape for selector
// parity tests. Mirrors the web's `SelectorFixture` interface
// (per IOS_MACOS_WIRE_FORMAT.md §5.1).
import Foundation

public struct SelectorFixture: Codable {
    public let name: String
    public let selector: String
    public let input: JSONValue
    public let expected: JSONValue
    public let seed: String?

    public init(name: String, selector: String, input: JSONValue, expected: JSONValue, seed: String? = nil) {
        self.name = name
        self.selector = selector
        self.input = input
        self.expected = expected
        self.seed = seed
    }
}

/// A dynamic JSON value (used for fixture inputs and expected
/// outputs, which are typed only at parity-test time).
public enum JSONValue: Codable, Equatable, Sendable {
    case null
    case bool(Bool)
    case int(Int)
    case double(Double)
    case string(String)
    case array([JSONValue])
    case object([String: JSONValue])

    public init(from decoder: Decoder) throws {
        let container = try decoder.singleValueContainer()
        if container.decodeNil() {
            self = .null
        } else if let b = try? container.decode(Bool.self) {
            self = .bool(b)
        } else if let i = try? container.decode(Int.self) {
            self = .int(i)
        } else if let d = try? container.decode(Double.self) {
            self = .double(d)
        } else if let s = try? container.decode(String.self) {
            self = .string(s)
        } else if let a = try? container.decode([JSONValue].self) {
            self = .array(a)
        } else if let o = try? container.decode([String: JSONValue].self) {
            self = .object(o)
        } else {
            throw DecodingError.dataCorruptedError(
                in: container,
                debugDescription: "Unrecognized JSON value"
            )
        }
    }

    public func encode(to encoder: Encoder) throws {
        var container = encoder.singleValueContainer()
        switch self {
        case .null: try container.encodeNil()
        case .bool(let b): try container.encode(b)
        case .int(let i): try container.encode(i)
        case .double(let d): try container.encode(d)
        case .string(let s): try container.encode(s)
        case .array(let a): try container.encode(a)
        case .object(let o): try container.encode(o)
        }
    }

    /// Convert the JSON value to a Swift type T. Used in
    /// parity tests to coerce the fixture's input/expected
    /// into the specific types each selector expects.
    public func to<T: Decodable>(_ type: T.Type) throws -> T {
        let data = try JSONEncoder().encode(self)
        return try JSONDecoder().decode(type, from: data)
    }
}
```

## Task 3: Build the deterministic fixture source (`lib/select.fixtures.ts`) and wire both consumers

**Why this is a rewrite, not an "extend":** the original plan assumed `scripts/export-fixtures.ts`
could scrape a structured `CASES` array out of `lib/select.test.ts` (e.g. via an
`extractTestCases('select.test.ts')` step). That **cannot work**: `lib/select.test.ts` is **74 flat
`test('…', () => {…})` calls** with inline literals and **`Math.random()`-generated ids** — there is no
structured, deterministic source to extract, and re-running it would emit non-reproducible fixtures.
`scripts/export-fixtures.ts` also **does not exist yet** — it is a Phase 1.0 artifact **to be created**.

**New design (canonical §G):** introduce a standalone, deterministic **`frontend/lib/select.fixtures.ts`**
that exports `export const CASES: SelectorFixture[]` — fixed ids, **no `Math.random()`**, each case
`{ name, selector, input, expected, seed? }`. `CASES` is the **single source of truth**, consumed by
**both**:
1. `frontend/lib/select.test.ts` — iterates `CASES` and asserts each (the web-side oracle), and
2. `frontend/scripts/export-fixtures.ts` (the Phase 1.0 artifact) — imports `CASES` and serializes each
   case to the JSON the Swift parity target reads.

This decouples the fixtures from bun-test mechanics and guarantees the Swift side checks the **exact same**
expected values the web asserts.

**Files:**
- Create: `frontend/lib/select.fixtures.ts` (the deterministic `CASES` source of truth)
- Modify: `frontend/lib/select.test.ts` (refactor the 74 inline tests to iterate `CASES`)
- Modify: `frontend/scripts/export-fixtures.ts` (the Phase 1.0 script — import `CASES`, drop any scrape/`extractTestCases` step) to cover the 25 new Phase 1.5 selectors
- Create: 25 new fixture JSON files in `frontend/ios/FinchCore/Tests/Fixtures/selectors/`

```swift
extension SelectorParityTests {
    private func runParityTest(selector: String, file: StaticString = #file, line: UInt = #line) throws {
        // Find all fixtures for this selector
        let fixturesDir = Bundle.module.url(
            forResource: "Fixtures/selectors", withExtension: nil
        )!
        let files = try FileManager.default.contentsOfDirectory(at: fixturesDir, ...)
        let matchingFiles = files.filter { $0.lastPathComponent.hasPrefix("\(selector)__") && $0.pathExtension == "json" }
        XCTAssertFalse(matchingFiles.isEmpty, "No fixtures found for \(selector)")

        for file in matchingFiles {
            let data = try Data(contentsOf: file)
            let fixture = try JSONDecoder().decode(SelectorFixture.self, from: data)
            // Dispatch to the right selector
            let actual: Any
            switch selector {
            case "currentMonth":
                actual = try Selectors.currentMonth(
                    txns: fixture.input.to([Tx].self),
                    ledgerId: fixture.input["ledgerId"].to(String.self)
                )
            case "balanceSeries":
                actual = try Selectors.balanceSeries(
                    txns: fixture.input.to([Tx].self),
                    accountId: fixture.input["accountId"].to(String.self),
                    currentBalance: Decimal(fixture.input["currentBalance"].to(Double.self))
                )
            // (... 30 more cases — full dispatch table)
            default:
                XCTFail("Unknown selector \(selector)"); return
            }
            XCTAssertEqual(
                try JSONEncoder().encode(actual as Any),
                try JSONEncoder().encode(fixture.expected),
                "Selector \(selector), case \(fixture.name) mismatch"
            )
        }
    }
}
```

```swift
public extension Projection {
    /// Project the `holdings` table for the given ledger.
    static func holdings(dbPool: DatabasePool, ledgerId: String) throws -> [Holding] {
        try dbPool.read { db in
            try Holding.fetchAll(db, sql: """
                SELECT h.* FROM holdings h
                JOIN accounts a ON a.id = h.account_id
                WHERE a.ledger_id = ?
            """, arguments: [ledgerId])
        }
    }

    /// Project the `exchange_rates` table as a [currency: Decimal] map.
    static func exchangeRates(dbPool: DatabasePool) throws -> [String: Decimal] {
        try dbPool.read { db in
            let rows = try Row.fetchAll(db, sql: "SELECT currency, rate FROM exchange_rates")
            var rates: [String: Decimal] = [:]
            for row in rows {
                let currency: String = row["currency"] ?? ""
                let rate: Double = row["rate"] ?? 1.0
                rates[currency] = Decimal(rate)
            }
            return rates
        }
    }

    /// Project the row counts (used by the pack manifest's
    /// `db.row_counts` field).
    static func rowCounts(dbPool: DatabasePool) throws -> [String: Int] {
        try dbPool.read { db in
            let tables = [
                "entries", "postings", "accounts", "account_groups",
                "categories", "counterparties", "tags", "rules",
                "scheduled", "transfers", "holdings", "budgets",
                "budget_groups", "budget_history", "attachments",
                "exchange_rates"
            ]
            var counts: [String: Int] = [:]
            for table in tables {
                let count = try Int.fetchOne(
                    db, sql: "SELECT COUNT(*) FROM \(table)"
                ) ?? 0
                counts[table] = count
            }
            return counts
        }
    }
}
```
